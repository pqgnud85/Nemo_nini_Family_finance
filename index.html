    <script>

        // --- CÁC HÀM TIỆN ÍCH ---


        /**

         * Định dạng số thành chuỗi có dấu phẩy phân cách hàng nghìn.

         * @param {number} num - Số cần định dạng (triệu VNĐ).

         * @returns {string} Chuỗi đã định dạng.

         */

        const formatNumber = (num) => {

            if (num === null || num === undefined) return '0';

            return Math.round(num).toLocaleString('vi-VN');

        };


        /**

         * Đồng bộ hóa hai input tỷ lệ phần trăm (a + b = 100).

         * @param {string} changedId - ID của input vừa được thay đổi.

         * @param {string} targetId - ID của input cần cập nhật.

         */

        const syncPercentage = (changedId, targetId) => {

            const changedInput = document.getElementById(changedId);

            const targetInput = document.getElementById(targetId);


            let changedValue = parseFloat(changedInput.value) || 0;

            if (changedValue < 0) changedValue = 0;

            if (changedValue > 100) changedValue = 100;

            changedInput.value = changedValue;


            let targetValue = 100 - changedValue;

            targetInput.value = targetValue.toFixed(0);

        };


        /**

         * Lấy giá trị số từ một input, xử lý trường hợp giá trị rỗng/không hợp lệ.

         * @param {string} id - ID của input.

         * @returns {number} Giá trị số (hoặc 0 nếu không hợp lệ).

         */

        const getInputValue = (id) => {

            const element = document.getElementById(id);

            if (!element) return 0;

            const value = parseFloat(element.value);

            return isNaN(value) ? 0 : value;

        };


        /**

         * Điền các dropdown chọn năm.

         */

        const populateDropdowns = () => {

            const startYear = getInputValue('startYear');

            const duration = getInputValue('simulationDuration');

            const endYear = startYear + duration;


            const dropdownIds = [

                'sellLand1Year', 'sellLand2Year',

                'expenseChangeYear1', 'expenseChangeYear2'

            ];


            dropdownIds.forEach(id => {

                const select = document.getElementById(id);

                if (!select) return;


                // Xóa các option cũ

                select.innerHTML = '';


                // Thêm option "Không"

                const optionNone = document.createElement('option');

                optionNone.value = '0';

                optionNone.textContent = 'Không';

                select.appendChild(optionNone);


                // Thêm các năm mô phỏng

                for (let year = startYear; year <= endYear; year++) {

                    const option = document.createElement('option');

                    option.value = year;

                    option.textContent = year;

                    select.appendChild(option);

                }


                // Thiết lập giá trị mặc định cho các mốc

                if (id === 'sellLand1Year') select.value = startYear + 10;

                if (id === 'sellLand2Year') select.value = startYear + 20;

                if (id === 'expenseChangeYear1') select.value = startYear + 18;

                if (id === 'expenseChangeYear2') select.value = startYear + 22;

            });

        };


        // --- HÀM TÍNH TOÁN CHÍNH ---


        let financialChart; // Biến toàn cục để lưu trữ biểu đồ Chart.js


        const calculateAndRender = () => {

            // 1. LẤY THAM SỐ ĐẦU VÀO

            const startYear = getInputValue('startYear');

            const duration = getInputValue('simulationDuration');

            const inflationRate = getInputValue('inflationRate') / 100;

            const savingInterestRate = getInputValue('savingInterestRate') / 100;

            const stockInterestRate = getInputValue('stockInterestRate') / 100;

            const loanInterestRate = getInputValue('loanInterestRate') / 100;

            const salaryIncreaseRate = getInputValue('salaryIncrease') / 100;

            const land1Appreciation = getInputValue('land1Appreciation') / 100;

            const land2Appreciation = getInputValue('land2Appreciation') / 100;


            let initialSavings = getInputValue('initialSavings');

            let initialStocks = getInputValue('initialStocks');

            let initialLand1 = getInputValue('initialLand1');

            let initialLand2 = getInputValue('initialLand2');

            let initialApartment = getInputValue('initialApartment');

            let loanApartment = getInputValue('loanApartment');

            let initialMonthlyExpense = getInputValue('initialMonthlyExpense');

            let incomeWife = getInputValue('incomeWife');

            let incomeHusband = getInputValue('incomeHusband');

            let incomeRental = getInputValue('incomeRental');


            const surplusToSavingsRatio = getInputValue('surplusToSavings') / 100;

            const surplusToStocksRatio = getInputValue('surplusToStocks') / 100;

            const sellLand1ToSavingsRatio = getInputValue('sellLand1ToSavings') / 100;

            const sellLand1ToStocksRatio = getInputValue('sellLand1ToStocks') / 100;

            const sellLand2ToSavingsRatio = getInputValue('sellLand2ToSavings') / 100;

            const sellLand2ToStocksRatio = getInputValue('sellLand2ToStocks') / 100;


            // Sự kiện Mốc

            const sellLand1Year = getInputValue('sellLand1Year');

            const sellLand2Year = getInputValue('sellLand2Year');

            const expenseChangeYear1 = getInputValue('expenseChangeYear1');

            const expenseChangeAmount1 = getInputValue('expenseChangeAmount1');

            const expenseChangeYear2 = getInputValue('expenseChangeYear2');

            const expenseChangeAmount2 = getInputValue('expenseChangeAmount2');


            // 2. MÔ PHỎNG THEO TỪNG NĂM

            let results = [];

            let ffYear = 'Không đạt';

            let ffSafetyRatio = '-';


            // Biến trạng thái để theo dõi việc bán tài sản

            let land1Sold = false;

            let land2Sold = false;


            // Biến trạng thái Thu nhập & Chi tiêu theo năm

            let currentMonthlyExpense = initialMonthlyExpense;

            let currentIncomeWife = incomeWife;

            let currentIncomeHusband = incomeHusband;

            let currentLoan = loanApartment;



            for (let year = 0; year <= duration; year++) {

                const currentYear = startYear + year;

                let data = {

                    year: currentYear,

                    savings: 0,

                    stocks: 0,

                    land1: 0,

                    land2: 0,

                    apartment: 0,

                    loan: 0,

                    netWorth: 0,

                    annualExpense: 0,

                    ffGoal: 0,

                    isFF: false,

                    event: ''

                };


                // --- TÍNH TOÁN ĐẦU NĂM (Tích lũy năm trước) ---

                if (year > 0) {

                    const prev = results[year - 1];

                    let eventDescription = '';


                    // 1. Tăng trưởng tài sản có thanh khoản (Tiết kiệm, Chứng khoán)

                    data.savings = prev.savings * (1 + savingInterestRate * (1 - inflationRate)); // Điều chỉnh lạm phát cho Tiết kiệm (Không bắt buộc, nhưng hợp lý hơn)

                    data.stocks = prev.stocks * (1 + stockInterestRate * (1 - inflationRate)); // Điều chỉnh lạm phát cho Chứng khoán


                    // 2. Tăng trưởng tài sản vật chất (Đất, Chung cư)

                    data.land1 = prev.land1 * (1 + land1Appreciation);

                    data.land2 = prev.land2 * (1 + land2Appreciation);

                    data.apartment = prev.apartment * (1 + inflationRate); // Giả sử chung cư tăng trưởng bằng lạm phát


                    // 3. Tăng trưởng Chi tiêu & Lương

                    currentMonthlyExpense = prev.annualExpense / 12 * (1 + inflationRate); // Chi tiêu năm trước, tăng theo lạm phát

                    currentIncomeWife = currentIncomeWife * (1 + salaryIncreaseRate);

                    currentIncomeHusband = currentIncomeHusband * (1 + salaryIncreaseRate);


                    // 4. Trả nợ

                    // Giả định trả nợ gốc hàng năm (10% khoản vay ban đầu, hoặc trả hết nếu gần hết)

                    const annualPrincipalPayment = Math.min(loanApartment * 0.1, prev.loan);

                    currentLoan = Math.max(0, prev.loan - annualPrincipalPayment);

                    // Lãi suất trả: prev.loan * loanInterestRate (đã được tính vào Chi tiêu bên dưới)



                    // --- XỬ LÝ SỰ KIỆN MỐC TRONG NĂM ---


                    // 5. Sự kiện Bán Đất 1

                    if (currentYear === sellLand1Year && !land1Sold) {

                        const land1Value = data.land1;

                        data.savings += land1Value * sellLand1ToSavingsRatio;

                        data.stocks += land1Value * sellLand1ToStocksRatio;

                        data.land1 = 0;

                        land1Sold = true;

                        eventDescription += `Bán Đất 1 (Giá: ${formatNumber(land1Value)} tr). Phân bổ: Tiết kiệm ${sellLand1ToSavingsRatio * 100}%, CK ${sellLand1ToStocksRatio * 100}%. `;

                    }


                    // 6. Sự kiện Bán Đất 2

                    if (currentYear === sellLand2Year && !land2Sold) {

                        const land2Value = data.land2;

                        data.savings += land2Value * sellLand2ToSavingsRatio;

                        data.stocks += land2Value * sellLand2ToStocksRatio;

                        data.land2 = 0;

                        land2Sold = true;

                        eventDescription += `Bán Đất 2 (Giá: ${formatNumber(land2Value)} tr). Phân bổ: Tiết kiệm ${sellLand2ToSavingsRatio * 100}%, CK ${sellLand2ToStocksRatio * 100}%. `;

                    }


                    // 7. Sự kiện Tăng Chi tiêu (Lần 1)

                    if (currentYear === expenseChangeYear1) {

                        // Chi tiêu tăng thêm này cũng phải được điều chỉnh lạm phát sau 1 năm

                        const realIncrease = expenseChangeAmount1 * Math.pow(1 + inflationRate, year);

                        currentMonthlyExpense += realIncrease;

                        eventDescription += `Chi tiêu tăng thêm Lần 1: ${formatNumber(realIncrease)} tr/tháng. `;

                    }


                    // 8. Sự kiện Tăng Chi tiêu (Lần 2)

                    if (currentYear === expenseChangeYear2) {

                        // Chi tiêu tăng thêm này cũng phải được điều chỉnh lạm phát sau 1 năm

                        const realIncrease = expenseChangeAmount2 * Math.pow(1 + inflationRate, year);

                        currentMonthlyExpense += realIncrease;

                        eventDescription += `Chi tiêu tăng thêm Lần 2: ${formatNumber(realIncrease)} tr/tháng. `;

                    }


                    data.event = eventDescription.trim();


                } else {

                    // --- NĂM 0 (DỮ LIỆU BAN ĐẦU) ---

                    data.savings = initialSavings;

                    data.stocks = initialStocks;

                    data.land1 = initialLand1;

                    data.land2 = initialLand2;

                    data.apartment = initialApartment;

                    data.loan = loanApartment;

                    currentMonthlyExpense = initialMonthlyExpense;

                }


                // --- TÍNH TOÁN DÒNG TIỀN TRONG NĂM ---

                const totalMonthlyIncome = currentIncomeWife + currentIncomeHusband + incomeRental;

                const loanInterestMonthly = (data.loan > 0) ? (data.loan / 12 * loanInterestRate) : 0;


                // Tổng chi tiêu hàng tháng: Chi tiêu cơ bản + Lãi suất vay + Trả nợ gốc (đã tính ở bước 4)

                // Tuy nhiên, để đơn giản, chỉ cần tính: Chi tiêu cơ bản + Lãi suất vay

                const totalMonthlyExpense = currentMonthlyExpense + loanInterestMonthly;


                // Thặng dư hàng tháng

                const monthlySurplus = totalMonthlyIncome - totalMonthlyExpense;

                const annualSurplus = monthlySurplus * 12;


                // Tích lũy thêm từ thặng dư

                if (annualSurplus > 0) {

                    const surplusToSavings = annualSurplus * surplusToSavingsRatio;

                    const surplusToStocks = annualSurplus * surplusToStocksRatio;


                    data.savings += surplusToSavings;

                    data.stocks += surplusToStocks;

                }


                // --- TÍNH KẾT QUẢ CUỐI NĂM ---

                data.annualExpense = currentMonthlyExpense * 12;

                data.netWorth = data.savings + data.stocks + data.land1 + data.land2 + data.apartment - data.loan;


                // Mục tiêu Tự do Tài chính (25x Chi tiêu)

                data.ffGoal = data.annualExpense * 25;


                // Xác định Tự do Tài chính

                if (data.netWorth >= data.ffGoal) {

                    data.isFF = true;

                    if (ffYear === 'Không đạt') {

                        ffYear = currentYear;

                        // Tính tỷ lệ an toàn (Safety Ratio) khi đạt FF

                        ffSafetyRatio = (data.netWorth / data.annualExpense).toFixed(1);

                    }

                }


                results.push(data);

            }


            // 3. HIỂN THỊ KẾT QUẢ TỔNG QUAN


            const finalData = results[results.length - 1];

            document.getElementById('netWorthFinal').textContent = formatNumber(finalData.netWorth);

            document.getElementById('ffYear').textContent = ffYear;

            document.getElementById('ffSafetyRatio').textContent = `${ffSafetyRatio} x`;


            // 4. HIỂN THỊ BẢNG DỮ LIỆU

            const dataTableBody = document.getElementById('dataTableBody');

            dataTableBody.innerHTML = ''; // Xóa dữ liệu cũ


            results.forEach(row => {

                const tr = document.createElement('tr');

                tr.className = row.isFF ? 'bg-green-100 hover:bg-green-200 transition duration-100' : 'hover:bg-gray-50';

                tr.innerHTML = `

                    <td class="px-3 py-2 whitespace-nowrap text-sm font-medium ${row.isFF ? 'text-green-800' : 'text-gray-900'}">${row.year} ${row.isFF ? '✨' : ''}</td>

                    <td class="px-3 py-2 whitespace-nowrap text-sm text-gray-500 font-bold">${formatNumber(row.netWorth)}</td>

                    <td class="px-3 py-2 whitespace-nowrap text-sm text-gray-500">${formatNumber(row.annualExpense)}</td>

                    <td class="px-3 py-2 whitespace-nowrap text-sm text-indigo-600 font-semibold">${formatNumber(row.ffGoal)}</td>
